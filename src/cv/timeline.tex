        % Two neighbouring events less than ~5 months apart cannot both carry a
        % 45-degree label without overlapping, so such a pair shares one label
        % placed midway between the two dots. A two-line label (\\) is possible
        % when the previous event is more than a year away, since the extra line
        % stacks towards it.
        % Usage: \pairlabel{date1}{date2}{label}
        \newcommand{\pairlabel}[3]{%
            \pgfmathsetlength\xstart{((#1+#2)/2-\theyearstart)*\unit}%
            \node[eventlabel,align=left] at (\xstart,0) {#3};%
        }
        \newcommand{\sep}{{\,\textperiodcentered\,}}
